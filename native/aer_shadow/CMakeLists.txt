cmake_minimum_required(VERSION 3.21)
project(aer_shadow LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "aer_shadow is Windows-only; Linux keeps gnome-rdp")
endif()
find_package(WinPR 3.32 CONFIG REQUIRED)
find_package(FreeRDP 3.32 CONFIG REQUIRED)
find_package(FreeRDP-Server 3.32 CONFIG REQUIRED)
find_library(SHADOW_LIB NAMES freerdp-shadow3 REQUIRED)
find_path(CJSON_INCLUDE NAMES cjson/cJSON.h REQUIRED)
find_library(CJSON_LIB NAMES cjson REQUIRED)
add_executable(aer-shadow aer_shadow.cpp)
target_compile_features(aer-shadow PRIVATE cxx_std_17)
target_compile_definitions(aer-shadow PRIVATE UNICODE _UNICODE NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_include_directories(aer-shadow PRIVATE ${CJSON_INCLUDE})
target_link_libraries(aer-shadow PRIVATE ${SHADOW_LIB} freerdp-server freerdp winpr ${CJSON_LIB} d3d11 dxgi user32 advapi32)
target_compile_options(aer-shadow PRIVATE /W4 /we4013)
set_target_properties(aer-shadow PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
install(TARGETS aer-shadow RUNTIME_DEPENDENCY_SET shadow_deps RUNTIME DESTINATION .)
install(RUNTIME_DEPENDENCY_SET shadow_deps DESTINATION . DIRECTORIES "${CMAKE_PREFIX_PATH}/bin"
  PRE_EXCLUDE_REGEXES "^api-ms-" "^ext-ms-" "[Vv][Cc][Rr][Uu][Nn][Tt][Ii][Mm][Ee]" "[Mm][Ss][Vv][Cc][Pp]"
  POST_EXCLUDE_REGEXES "[Ss]ystem32")
set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
include(InstallRequiredSystemLibraries)
install(FILES ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS} DESTINATION .)
